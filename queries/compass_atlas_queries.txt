=============================================================
  COMPASS – AGGREGATION PIPELINE TAB
  (Add each stage one at a time in the Aggregation tab)
=============================================================

── CA1:  Group by Event_Type (count > 140), avg ticket, sorted
         [Shows: $group → $match → $project → $sort]

Stage 1 – $group
{
  "_id": "$Event_Type",
  "Total_Events": { "$sum": 1 },
  "Avg_Ticket": { "$avg": "$Ticket_Price" }
}

Stage 2 – $match
{ "Total_Events": { "$gt": 140 } }

Stage 3 – $project
{
  "_id": 1,
  "Total_Events": 1,
  "Avg_Ticket": { "$round": ["$Avg_Ticket", 2] }
}

Stage 4 – $sort
{ "Avg_Ticket": -1 }

─────────────────────────────────────────────────────────────

── CA2:  $bucket on Ticket_Price ranges with avg crowd
         [Shows: $bucket]

Stage 1 – $bucket
{
  "groupBy": "$Ticket_Price",
  "boundaries": [0, 1, 31, 61, 91, 121],
  "default": "Other",
  "output": {
    "Event_Count": { "$sum": 1 },
    "Avg_Crowd":   { "$avg": "$Actual_Crowd" }
  }
}

Stage 2 – $project
{
  "_id": 1,
  "Event_Count": 1,
  "Avg_Crowd": { "$round": ["$Avg_Crowd", 0] }
}

─────────────────────────────────────────────────────────────

── CA3:  $unwind Sponsors array, group by sponsor, top 5
         [Shows: $unwind → $group → $project → $sort → $limit]

Stage 1 – $unwind
"$Sponsors"

Stage 2 – $group
{
  "_id": "$Sponsors",
  "Total_Events":  { "$sum": 1 },
  "Avg_SponValue": { "$avg": "$Sponsorship_Value" }
}

Stage 3 – $project
{
  "_id": 1,
  "Total_Events": 1,
  "Avg_SponValue": { "$round": ["$Avg_SponValue", 0] }
}

Stage 4 – $sort
{ "Total_Events": -1 }

Stage 5 – $limit
5

─────────────────────────────────────────────────────────────

── CA4:  $facet multi-faced dashboard
         [Shows: $facet with 3 sub-pipelines]

Stage 1 – $facet
{
  "TopVenuesByCrowd": [
    { "$group": { "_id": "$Sports_Hall.Name", "Total_Crowd": { "$sum": "$Actual_Crowd" }, "Total_Events": { "$sum": 1 } } },
    { "$sort": { "Total_Crowd": -1 } },
    { "$limit": 5 }
  ],
  "AtmosphereBreakdown": [
    { "$group": { "_id": "$Atmosphere", "Total_Events": { "$sum": 1 } } },
    { "$sort": { "Total_Events": -1 } }
  ],
  "AvgTicketByType": [
    { "$group": { "_id": "$Event_Type", "Avg_Ticket": { "$avg": "$Ticket_Price" } } },
    { "$project": { "_id": 1, "Avg_Ticket": { "$round": ["$Avg_Ticket", 2] } } },
    { "$sort": { "Avg_Ticket": -1 } }
  ]
}

─────────────────────────────────────────────────────────────

── CA5:  Computed field: total revenue per event (Concert/Sport 2022)
         [Shows: $match → $addFields → $match → $sort]

Stage 1 – $match
{ "Event_Type": { "$in": ["Concert", "Sport"] }, "Event_Year": 2022 }

Stage 2 – $addFields
{
  "Total_Revenue": {
    "$add": [
      "$Food_Sales",
      "$Merchandise_Sales",
      { "$multiply": ["$Ticket_Price", "$Actual_Crowd"] }
    ]
  }
}

Stage 3 – $match
{ "Total_Revenue": { "$gt": 400000 } }

Stage 4 – $sort
{ "Total_Revenue": -1 }
